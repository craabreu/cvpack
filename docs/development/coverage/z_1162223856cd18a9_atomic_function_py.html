21class AtomicFunction(BaseCustomFunction, openmm.CustomCompoundBondForce): 

22 r""" 

23 A generic function of the coordinates of :math:`m` groups of :math:`n` atoms: 

24 

25 .. math:: 

26 

27 f({\bf r}) = \sum_{i=1}^m F\left( 

28 {\bf r}_{i,1}, {\bf r}_{i,2}, \dots, {\bf r}_{i,n} 

29 \right) 

30 

31 where :math:`F` is a user-defined function and :math:`{\bf r}_{i,j}` is the 

32 coordinate of the :math:`j`-th atom of the :math:`i`-th group. 

33 

34 The function :math:`F` is defined as a string and can be any expression supported 

35 by :OpenMM:`CustomCompoundBondForce`. If the expression contains named parameters, 

36 the value of each parameter can be passed in one of three ways: 

37 

38 #. By a semicolon-separated definition in the function string, such as described 

39 in the :OpenMM:`CustomCompoundBondForce` documentation. In this case, the 

40 parameter value will be the same for all groups of atoms. 

41 

42 #. By a 1D array or list of length :math:`m` passed as a keyword argument to 

43 the :class:`AtomicFunction` constructor. In this case, each value will be 

44 assigned to a different group of atoms. 

45 

46 #. By a scalar passed as a keyword argument to the :class:`AtomicFunction` 

47 constructor. In this case, the parameter will apply to all atom groups and will 

48 become available for on-the-fly modification during a simulation via the 

49 ``setParameter`` method of an :OpenMM:`Context` object. **Warning**: other 

50 collective variables or :OpenMM:`Force` objects in the same system will share 

51 the same values of equal-named parameters. 

52 

53 Parameters 

54 ---------- 

55 function 

56 The function to be evaluated. It must be a valid 

57 :OpenMM:`CustomCompoundBondForce` expression. 

58 groups 

59 The indices of the atoms in each group, passed as a 2D array-like object of 

60 shape `(m, n)`, where `m` is the number of groups and `n` is the number of 

61 atoms per group. If a 1D object is passed, it is assumed that `m` is 1 and 

62 `n` is the length of the object. 

63 unit 

64 The unit of measurement of the collective variable. It must be compatible 

65 with the MD unit system (mass in `daltons`, distance in `nanometers`, time 

66 in `picoseconds`, temperature in `kelvin`, energy in `kilojoules_per_mol`, 

67 angle in `radians`). If the collective variables does not have a unit, use 

68 `unit.dimensionless`. 

69 periodicBounds 

70 The lower and upper bounds of the collective variable if it is periodic, or 

71 ``None`` if it is not. 

72 pbc 

73 Whether to use periodic boundary conditions when computing atomic distances. 

74 name 

75 The name of the collective variable. 

76 

77 Keyword Args 

78 ------------ 

79 **parameters 

80 The named parameters of the function. Each parameter can be a 1D array-like 

81 object or a scalar. In the former case, the array must have length :math:`m` 

82 and each entry will be assigned to a different group of atoms. In the latter 

83 case, it will define a global :OpenMM:`Context` parameter. 

84 

85 Raises 

86 ------ 

87 ValueError 

88 If the groups are not specified as a 1D or 2D array-like object. 

89 ValueError 

90 If the unit of the collective variable is not compatible with the MD unit 

91 system. 

92 

93 Example 

94 ------- 

95 >>> import cvpack 

96 >>> import openmm 

97 >>> import numpy as np 

98 >>> from openmm import unit 

99 >>> from openmmtools import testsystems 

100 >>> model = testsystems.AlanineDipeptideVacuum() 

101 >>> angle1 = cvpack.Angle(0, 5, 10) 

102 >>> angle2 = cvpack.Angle(10, 15, 20) 

103 >>> colvar = cvpack.AtomicFunction( 

104 ... "(kappa/2)*(angle(p1, p2, p3) - theta0)^2", 

105 ... unit.kilojoules_per_mole, 

106 ... [[0, 5, 10], [10, 15, 20]], 

107 ... kappa = 1000 * unit.kilojoules_per_mole/unit.radian**2, 

108 ... theta0 = [np.pi/2, np.pi/3] * unit.radian, 

109 ... ) 

110 >>> for cv in [angle1, angle2, colvar]: 

111 ... cv.addToSystem(model.system) 

112 >>> integrator = openmm.VerletIntegrator(0) 

113 >>> platform = openmm.Platform.getPlatformByName('Reference') 

114 >>> context = openmm.Context(model.system, integrator, platform) 

115 >>> context.setPositions(model.positions) 

116 >>> theta1 = angle1.getValue(context) / openmm.unit.radian 

117 >>> theta2 = angle2.getValue(context) / openmm.unit.radian 

118 >>> 500*((theta1 - np.pi/2)**2 + (theta2 - np.pi/3)**2) 

119 429.479... 

120 >>> colvar.getValue(context) 

121 429.479... kJ/mol 

122 >>> context.setParameter( 

123 ... "kappa", 

124 ... 2000 * unit.kilojoules_per_mole/unit.radian**2 

125 ... ) 

126 >>> colvar.getValue(context) 

127 858.958... kJ/mol 

128 """

163 def _fromCustomForce( # noqa: PLR0912 

164 cls, 

165 force: openmm.CustomAngleForce 

166 | openmm.CustomBondForce 

167 | openmm.CustomCompoundBondForce 

168 | openmm.CustomExternalForce 

169 | openmm.CustomTorsionForce, 

170 unit: mmunit.Unit, 

171 periodicBounds: VectorQuantity | None = None, 

172 pbc: bool = False, 

173 ) -> AtomicFunction: 

174 """ 

175 Create a :class:`AtomicFunction` from an object of :openmm:`CustomBondForce`, 

176 :openmm:`CustomAngleForce`, :openmm:`CustomTorsionForce`, or 

177 :openmm:`CustomCompoundBondForce` class. 

178 """ 

179 if isinstance(force, openmm.CustomExternalForce): 

180 number, item, definition = 1, "Particle", "; x=x1; y=y1; z=z1" 

181 elif isinstance(force, openmm.CustomBondForce): 

182 number, item, definition = 2, "Bond", "; r=distance(p1, p2)" 

183 elif isinstance(force, openmm.CustomAngleForce): 

184 number, item, definition = 3, "Angle", "; theta=angle(p1, p2, p3)" 

185 elif isinstance(force, openmm.CustomTorsionForce): 

186 number, item, definition = 4, "Torsion", "; theta=dihedral(p1, p2, p3, p4)" 

187 elif isinstance(force, openmm.CustomCompoundBondForce): 

188 number, item, definition = force.getNumParticlesPerBond(), "Bond", "" 

189 else: 

190 raise TypeError(f"Unsupported force type: {type(force).__name__}") 

191 function = force.getEnergyFunction() + definition 

192 parameters = {} 

193 for i in range(force.getNumGlobalParameters()): 

194 name = force.getGlobalParameterName(i) 

195 value = force.getGlobalParameterDefaultValue(i) 

196 parameters[name] = value 

197 per_item_parameter_names = [] 

198 for i in range(getattr(force, f"getNumPer{item}Parameters")()): 

199 per_item_parameter_names.append( 

200 getattr(force, f"getPer{item}ParameterName")(i) 

201 ) 

202 for name in per_item_parameter_names: 

203 parameters[name] = [] 

204 atoms = [] 

205 for i in range(getattr(force, f"getNum{item}s")()): 

206 if isinstance(force, openmm.CustomCompoundBondForce): 

207 indices, per_item_parameters = force.getBondParameters(i) 

208 else: 

209 *indices, per_item_parameters = getattr( 

210 force, 

211 f"get{item}Parameters", 

212 )(i) 

213 atoms.append(indices) 

214 for name, value in zip(per_item_parameter_names, per_item_parameters): 

215 parameters[name].append(value) 

216 atoms = np.asarray(atoms).reshape(-1, number) 

217 return cls(function, unit, atoms, periodicBounds, pbc, **parameters)

227 Create a :class:`AtomicFunction` from an :OpenMM:`HarmonicBondForce`. 

228 """ 

229 parameters = {"r0": [], "k": []} 

230 atoms = [] 

231 for i in range(force.getNumBonds()): 

232 *indices, length, k = force.getBondParameters(i) 

233 atoms.append(indices) 

234 parameters["r0"].append(length) 

235 parameters["k"].append(k) 

236 return cls( 

237 "(k/2)*(distance(p1, p2)-r0)^2", 

238 unit, 

239 atoms, 

240 None, 

241 pbc, 

242 **parameters,

298 @classmethod 

299 def fromOpenMMForce( 

300 cls, 

301 force: openmm.Force,

303 periodicBounds: VectorQuantity | None = None, 

304 pbc: bool = False, 

305 ) -> AtomicFunction: 

306 """ 

307 Create an :class:`AtomicFunction` from an :OpenMM:`Force`. 

308 

309 Parameters 

310 ---------- 

311 force 

312 The force to be converted 

313 unit 

314 The unit of measurement of the collective variable. It must be 

315 compatible with the MD unit system (mass in `daltons`, distance in 

316 `nanometers`, time in `picoseconds`, temperature in `kelvin`, energy in 

317 `kilojoules_per_mol`, angle in `radians`). If the collective variables 

318 does not have a unit, use `unit.dimensionless`. 

319 periodicBounds 

320 The lower and upper bounds of the collective variable if it is periodic, 

321 or ``None`` if it is not. This parameter is considered only if `force` is 

322 a custom :OpenMM:`Force`. 

323 pbc 

324 Whether to use periodic boundary conditions when computing atomic 

325 distances. 

326 

327 Raises 

328 ------ 

329 TypeError 

330 If the force is not convertible to an :class:`AtomicFunction` 

331 

332 Examples 

333 -------- 

334 >>> import cvpack 

335 >>> import numpy as np 

336 >>> import openmm 

337 >>> from openmm import unit 

338 >>> from openmm import app 

339 >>> from openmmtools import testsystems 

340 >>> model = testsystems.LysozymeImplicit() 

341 >>> residues = [r for r in model.topology.residues() if 59 <= r.index <= 79] 

342 >>> helix_content = cvpack.HelixTorsionContent(residues) 

343 >>> model.system.addForce(helix_content) 

344 6 

345 >>> num_atoms = model.system.getNumParticles() 

346 >>> mean_x = openmm.CustomExternalForce("x/num_atoms") 

347 >>> mean_x.addGlobalParameter("num_atoms", num_atoms) 

348 0 

349 >>> for i in range(num_atoms): 

350 ... _ = mean_x.addParticle(i, []) 

351 >>> model.system.addForce(mean_x) 

352 7 

353 >>> forces = {force.getName(): force for force in model.system.getForces()} 

354 >>> copies = { 

355 ... name: cvpack.AtomicFunction.fromOpenMMForce( 

356 ... force, unit.kilojoules_per_mole 

357 ... ) 

358 ... for name, force in forces.items() 

359 ... if name in [ 

360 ... "HarmonicBondForce", 

361 ... "HarmonicAngleForce", 

362 ... "PeriodicTorsionForce", 

363 ... "CustomExternalForce" 

364 ... ] 

365 ... } 

366 >>> copies["HelixTorsionContent"] = cvpack.AtomicFunction.fromOpenMMForce( 

367 ... helix_content, unit.dimensionless 

368 ... ) 

369 >>> indices = {} 

370 >>> for index, (name, force) in enumerate(copies.items(), start=1): 

371 ... _ = model.system.addForce(force) 

372 ... force.setForceGroup(index) 

373 ... indices[name] = index 

374 >>> platform = openmm.Platform.getPlatformByName('Reference') 

375 >>> integrator = openmm.VerletIntegrator(0) 

376 >>> context = openmm.Context(model.system, integrator, platform) 

377 >>> context.setPositions(model.positions) 

378 >>> for name in copies: 

379 ... state = context.getState(getEnergy=True, groups={indices[name]}) 

380 ... value = state.getPotentialEnergy() / unit.kilojoules_per_mole 

381 ... copy_value = copies[name].getValue(context) 

382 ... print(f"{name}: original={value:.6f}, copy={copy_value}") 

383 HarmonicBondForce: original=2094.312..., copy=2094.312... kJ/mol 

384 HarmonicAngleForce: original=3239.795..., copy=3239.795... kJ/mol 

385 PeriodicTorsionForce: original=4226.05..., copy=4226.05... kJ/mol 

386 CustomExternalForce: original=5.02155..., copy=5.02155... kJ/mol 

387 HelixTorsionContent: original=17.4528..., copy=17.4528... dimensionless 

388 """ 

389 if isinstance( 

390 force, 

391 ( 

392 openmm.CustomAngleForce, 

393 openmm.CustomBondForce, 

394 openmm.CustomCompoundBondForce, 

395 openmm.CustomExternalForce, 

396 openmm.CustomTorsionForce, 

397 ), 

398 ): 

399 return cls._fromCustomForce(force, unit, periodicBounds, pbc) 

400 if isinstance(force, openmm.HarmonicBondForce): 

401 return cls._fromHarmonicBondForce(force, unit, pbc) 

402 if isinstance(force, openmm.HarmonicAngleForce): 

403 return cls._fromHarmonicAngleForce(force, unit, pbc) 

404 if isinstance(force, openmm.PeriodicTorsionForce): 

405 return cls._fromPeriodicTorsionForce(force, unit, pbc) 

406 raise TypeError(f"Force {force} is not convertible to an AtomicFunction")
